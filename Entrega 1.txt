def Graficar_bbox(dict_anotaciones, path):

    imagen= imread(path)
    id_imagen= None
    nombre_imagen = os.path.basename(path)
    for img in dict_anotaciones['images']:
        if img['file_name']==nombre_imagen:
            id_imagen=img['id']
    
    bbox=None
    for anotacion in dict_anotaciones['annotations']:
        if anotacion['image_id']==id_imagen and anotacion['category_id']==3:
            bbox= anotacion['bbox']
    
    x = int(bbox[0])
    y = int(bbox[1])
    w = int(bbox[2])
    h = int(bbox[3])
    
    img_bbox= cv2.rectangle(imagen,(x,y),(x+w,y+h),(255,0,0),2)
            
    
    return img_bbox,bbox
